rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Allow user to access their own user document
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Allow user to access subcollections under their user document
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Allow access to public cigar catalog
    match /cigarCatalog/{document=**} {
      allow read: if request.auth != null;
    }

    // Allow access to recommendations
    match /recommendations/{document=**} {
      allow read: if request.auth != null;
    }

    // Allow creation of pending cigars
    match /pendingCigars/{document} {
      allow create: if request.auth != null;
      allow read: if request.auth != null && resource.data.submittedBy == request.auth.uid;
    }

    // ✅ logsMomentsImages collection
    match /logsMomentsImages/{docId} {
      allow create: if request.auth != null; // Any authenticated user can upload
      allow read: if resource.data.isPublic == true || request.auth.uid == resource.data.ownerId;
      allow update, delete: if request.auth.uid == resource.data.ownerId; // Only owner can update/delete
    }

    // ✅ postReactions collection for like/dislike functionality
    match /postReactions/{postId} {
      allow read: if request.auth != null; // Any authenticated user can read reactions
      allow write: if request.auth != null; // Any authenticated user can like/dislike
    }

    // ✅ NEW: Comments on posts
    // Path: /postComments/{postId}/comments/{commentId}
    match /postComments/{postId}/comments/{commentId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;

      // Only the author can update/delete their own comment
      allow update, delete: if request.auth != null
        && resource.data.userId == request.auth.uid;
    }
  }
}
